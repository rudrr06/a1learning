    counts[ip] = counts.get(ip, 0) + 1 adds one for that IP.
    def failed_logins(lines):
    counts = {}
    for line in lines:
        status, ip = line.split()
        if status == "FAIL":
            counts[ip] = counts.get(ip, 0) + 1
    return counts
